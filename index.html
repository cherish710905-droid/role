📊 구글 스프레드시트 (Apps Script) 연동

구글 스프레드시트의 확장 프로그램 > Apps Script를 웹 앱(Web App)으로 배포한 URL을 입력하면, 실시간으로 명단을 가져오거나 배정 결과를 원클릭으로 저장할 수 있습니다.

배포 시 '액세스 권한: 모든 사용자(Anyone)'로 설정된 Web App URL을 입력하세요.

💡 연동 3단계 가이드

  1. 구글 드라이브에서 스프레드시트를 만들고 A열에 학생 이름, B열에 역할명, C열에 역할 설명을 입력합니다.
  2. 상단 메뉴 [확장 프로그램] > [Apps Script]를 클릭하고 아래 템플릿 코드를 붙여넣습니다.
  3. 우측 상단 [배포] > [새 배포] > 유형: 웹 앱 선택 후 생성된 URL을 위에 붙여넣으면 완료!
Apps Script 소스 코드 템플릿 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push({ name: data[i][1], duty: data[i][2] || '' });
  }
  return ContentService.createTextOutput(JSON.stringify({
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  sheet.getRange(1, 5, 1, 3).setValues([['배정 학생', '역할명', '배정 일시']]);
  var rows = payload.assignments.map(function(item) {
    return [item.student, item.roleTitle, new Date()];
  });
  if (rows.length > 0) {
    sheet.getRange(2, 5, rows.length, 3).setValues(rows);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
